restrained 1
vhroot $SERVER_ROOT/$VH_NAME/
enablescript 1
chrootmode 0
allowsymbollink 1
configfile $SERVER_ROOT/conf/vhosts/$VH_NAME/vhconf.conf
setuidmode 2

virtualhostconfig  {
    docroot $VH_ROOT/public/
    enablegzip 1

    rewrite  {
        enable 0
        loglevel 0
    }

    accesslog $SERVER_ROOT/logs/$VH_NAME.access.log {
        compressarchive 0
        useserver 0
        keepdays 30
        logheaders 3
        rollingsize 500M
    }

    hotlinkctrl  {
        onlyself 1
        enablehotlinkctrl 0
        suffixes gif, jpeg, jpg
        allowdirectaccess 1
    }

    expires  {
        enableexpires 1
    }

    awstats  {
        workingdir $VH_ROOT/awstats
        awstatsuri /awstats/
        sitedomain localhost
        sitealiases 127.0.0.1 localhost
        updatemode 0
        updateinterval 86400
        updateoffset 0
        securedconn 0
    }

    index  {
        autoindexuri /_autoindex/default.php
        useserver 0
        autoindex 0
        indexfiles index.html
    }

    accesscontrol  {
        allow *
    }

    context / {
        railsenv 1
        maxconns 5
        location $VH_ROOT/
        type rails
        accesscontrol 
        adddefaultcharset off

        rewrite  {
            rules rewritefile .htaccess
            unknownkeywords rewritefile .htaccess
        }
    }

    errorlog  {
        useserver 1
    }
}
