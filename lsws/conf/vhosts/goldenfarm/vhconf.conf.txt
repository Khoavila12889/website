    loglevel WARN
    rollingsize 10M
    keepdays 7
}

    logformat "%h %l %u %t \"%r\" %>s %b"

    keepdays 7

    indexfiles index.php, index.html
}

    add lsapi:lsphp php
}

    type lsapi
    address uds://tmp/lshttpd/lsphp.sock
    maxconns 40
    env PHP_LSAPI_CHILDREN=35
    env LSAPI_AVOID_FORK=0
    env LSAPI_EXTRA_HEADERS=1
    env LSAPI_MAX_IDLE=30
    env LSAPI_MAX_IDLE_CHILDREN=5
    inittimeout 60
    retrytimeout 0
    persistconn 1
    respbuffer 0
    autostart 2
    path /usr/local/lsws/lsphp84/bin/lsphp
    backlog 300
    instances 1
    memsoftlimit 2684354560
    memhardlimit 3221225472
}

    enable 1
    autoloadhtaccess 1
